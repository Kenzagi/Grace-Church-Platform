    81→function EventList({ events, onSelect, onRemind }: { events: EventItem[]; onSelect: (e: EventItem) => void; onRemind: (e: EventItem) => void }) {
    82→  if (events.length === 0) return <EmptyState title="No events" description="Create a new event to get started." />;
    83→  const now = new Date();
    84→  const upcoming = events.filter((e) => new Date(e.startTime) >= now);
    85→  const past = events.filter((e) => new Date(e.startTime) < now).reverse();
    86→
    87→  return (
    88→    <div className="space-y-6">
    89→      {upcoming.length > 0 && (
    90→        <div>
    91→          <h3 className="text-sm font-semibold text-muted-foreground uppercase tracking-wide mb-3">Upcoming</h3>
    92→          <div className="space-y-2">
    93→            {upcoming.map((e) => <EventRow key={e.id} event={e} onSelect={onSelect} onRemind={onRemind} />)}
    94→          </div>
    95→        </div>
    96→      )}
    97→      {past.length > 0 && (
    98→        <div>
    99→          <h3 className="text-sm font-semibold text-muted-foreground uppercase tracking-wide mb-3">Past</h3>
   100→          <div className="space-y-2">
   101→            {past.map((e) => <EventRow key={e.id} event={e} onSelect={onSelect} onRemind={onRemind} />)}
   102→          </div>
   103→        </div>
   104→      )}
   105→    </div>
   106→  );
   107→}

   142→function CalendarView({ events, onSelect }: { events: EventItem[]; onSelect: (e: EventItem) => void }) {
   143→  const [month, setMonth] = React.useState(new Date());
   144→  const firstDay = new Date(month.getFullYear(), month.getMonth(), 1);
   145→  const lastDay = new Date(month.getFullYear(), month.getMonth() + 1, 0);
   146→  const startWeekday = firstDay.getDay(); // 0 = Sun
   147→  const daysInMonth = lastDay.getDate();
   148→
   149→  const days: (Date | null)[] = [];
   150→  for (let i = 0; i < startWeekday; i++) days.push(null);
   151→  for (let d = 1; d <= daysInMonth; d++) days.push(new Date(month.getFullYear(), month.getMonth(), d));
   152→
   153→  const eventsByDay = React.useMemo(() => {
   154→    const map: Record<string, EventItem[]> = {};
   155→    for (const e of events) {
   156→      const key = new Date(e.startTime).toDateString();
   157→      (map[key] = map[key] || []).push(e);
   158→    }
   159→    return map;
   160→  }, [events]);
   161→
   162→  return (
   163→    <SectionCard className="p-4">
   164→      <div className="flex items-center justify-between mb-4">
   165→        <h3 className="text-title text-primary">{month.toLocaleDateString("en-GB", { month: "long", year: "numeric" })}</h3>
   166→        <div className="flex items-center gap-1">
   167→          <Button variant="ghost" size="sm" onClick={() => setMonth(new Date(month.getFullYear(), month.getMonth() - 1, 1))}>
   168→            <Icon as={ChevronLeft} boundingBox={false} size={16} />
   169→          </Button>
   170→          <Button variant="ghost" size="sm" onClick={() => setMonth(new Date())}>Today</Button>
   171→          <Button variant="ghost" size="sm" onClick={() => setMonth(new Date(month.getFullYear(), month.getMonth() + 1, 1))}>
   172→            <Icon as={ChevronRight} boundingBox={false} size={16} />
   173→          </Button>
   174→        </div>
   175→      </div>
   176→      <div className="grid grid-cols-7 gap-1 mb-2">
   177→        {["Sun","Mon","Tue","Wed","Thu","Fri","Sat"].map((d) => (
   178→          <div key={d} className="text-center text-[11px] font-medium text-muted-foreground uppercase py-2">{d}</div>
   179→        ))}
   180→      </div>
   181→      <div className="grid grid-cols-7 gap-1">
   182→        {days.map((d, i) => {
   183→          if (!d) return <div key={i} className="min-h-20 rounded-sm bg-surface-alt/50" />;
   184→          const dayEvents = eventsByDay[d.toDateString()] || [];
   185→          const isToday = d.toDateString() === new Date().toDateString();
   186→          return (
   187→            <div key={i} className={cn("min-h-20 rounded-sm border border-border p-1.5", isToday && "ring-2 ring-primary ring-offset-1")}>
   188→              <p className={cn("text-xs font-medium mb-1", isToday ? "text-primary" : "text-muted-foreground")}>{d.getDate()}</p>
   189→              <div className="space-y-1">
   190→                {dayEvents.slice(0, 3).map((e) => (
   191→                  <button key={e.id} onClick={() => onSelect(e)} className="w-full text-left text-[11px] px-1.5 py-0.5 rounded-sm bg-primary text-primary-foreground truncate hover:opacity-80">
   192→                    {formatTime(e.startTime)} {e.title}
   193→                  </button>
   194→                ))}
   195→                {dayEvents.length > 3 && <p className="text-[10px] text-muted-foreground px-1">+{dayEvents.length - 3} more</p>}
   196→              </div>
   197→            </div>
   198→          );
   199→        })}
   200→      </div>
   201→    </SectionCard>
   202→  );
   203→}

   205→function AddEventDialog({ onClose, onSaved }: { onClose: () => void; onSaved: () => void }) {
   206→  const [form, setForm] = React.useState({
   207→    title: "", description: "", category: "service", location: "",
   208→    startTime: new Date(Date.now() + 86400000).toISOString().slice(0, 16),
   209→    endTime: "", capacity: "",
   210→  });
   211→  const [saving, setSaving] = React.useState(false);
   212→
   213→  async function handleSubmit() {
   214→    if (!form.title) { toast.error("Title required"); return; }
   215→    setSaving(true);
   216→    const res = await apiRequest("/api/events", { method: "POST", body: { ...form, endTime: form.endTime || null, capacity: form.capacity ? parseInt(form.capacity) : null } });
   217→    setSaving(false);
   218→    if (res.ok) { toast.success("Event created"); onSaved(); }
   219→    else toast.error(res.error || "Failed");
   220→  }
   221→
   222→  return (
   223→    <Dialog open onOpenChange={onClose}>
   224→      <DialogContent className="max-w-lg">
   225→        <DialogHeader><DialogTitle>New event</DialogTitle><DialogDescription>Schedule a service, meeting, or activity.</DialogDescription></DialogHeader>
   226→        <div className="space-y-4 py-2">
   227→          <div><Label className="text-sm font-medium mb-2 block">Title *</Label><Input value={form.title} onChange={(e) => setForm({ ...form, title: e.target.value })} /></div>
   228→          <div><Label className="text-sm font-medium mb-2 block">Description</Label><Textarea value={form.description} onChange={(e) => setForm({ ...form, description: e.target.value })} rows={2} /></div>
   229→          <div className="grid grid-cols-2 gap-4">
   230→            <div><Label className="text-sm font-medium mb-2 block">Category</Label>
   231→              <Select value={form.category} onValueChange={(v) => setForm({ ...form, category: v })}>
   232→                <SelectTrigger><SelectValue /></SelectTrigger>
   233→                <SelectContent>
   234→                  <SelectItem value="service">Service</SelectItem>
   235→                  <SelectItem value="meeting">Meeting</SelectItem>
   236→                  <SelectItem value="activity">Activity</SelectItem>
   237→                  <SelectItem value="group">Group</SelectItem>
   238→                </SelectContent>
   239→              </Select>
   240→            </div>
   241→            <div><Label className="text-sm font-medium mb-2 block">Location</Label><Input value={form.location} onChange={(e) => setForm({ ...form, location: e.target.value })} /></div>
   242→            <div><Label className="text-sm font-medium mb-2 block">Starts *</Label><Input type="datetime-local" value={form.startTime} onChange={(e) => setForm({ ...form, startTime: e.target.value })} /></div>
   243→            <div><Label className="text-sm font-medium mb-2 block">Ends</Label><Input type="datetime-local" value={form.endTime} onChange={(e) => setForm({ ...form, endTime: e.target.value })} /></div>
   244→            <div><Label className="text-sm font-medium mb-2 block">Capacity</Label><Input type="number" value={form.capacity} onChange={(e) => setForm({ ...form, capacity: e.target.value })} /></div>
   245→          </div>
   246→        </div>
   247→        <DialogFooter>
   248→          <Button variant="outline" size="sm" onClick={onClose}>Cancel</Button>
   249→          <Button size="sm" onClick={handleSubmit} disabled={saving}>{saving ? "Creating…" : "Create event"}</Button>
   250→        </DialogFooter>
   251→      </DialogContent>
   252→    </Dialog>
   253→  );
   254→}

   256→function EventDetailDialog({ event, onClose, onRemind }: { event: EventItem; onClose: () => void; onRemind: () => void }) {
   257→  const [deleting, setDeleting] = React.useState(false);
   258→  async function handleDelete() {
   259→    const res = await apiRequest(`/api/events/${event.id}`, { method: "DELETE" });
   260→    if (res.ok) { toast.success("Event deleted"); onClose(); }
   261→    else toast.error(res.error || "Failed");
   262→  }
   263→  const isUpcoming = new Date(event.startTime) >= new Date();
   264→  return (
   265→    <Dialog open onOpenChange={onClose}>
   266→      <DialogContent className="max-w-lg">
   267→        <DialogHeader>
   268→          <DialogTitle>{event.title}</DialogTitle>
   269→          <DialogDescription>{formatDateTime(event.startTime)}</DialogDescription>
   270→        </DialogHeader>
   271→        <div className="space-y-3 py-2">
   272→          {event.description && <p className="text-sm text-foreground">{event.description}</p>}
   273→          <dl className="grid grid-cols-2 gap-3 text-sm">
   274→            <div><dt className="text-caption">Category</dt><dd className="text-foreground capitalize">{event.category}</dd></div>
   275→            {event.location && <div><dt className="text-caption">Location</dt><dd className="text-foreground">{event.location}</dd></div>}
   276→            {event.endTime && <div><dt className="text-caption">Ends</dt><dd className="text-foreground">{formatTime(event.endTime)}</dd></div>}
   277→            {event._count?.attendance !== undefined && <div><dt className="text-caption">Attendance</dt><dd className="text-foreground">{event._count.attendance} recorded</dd></div>}
   278→          </dl>
   279→        </div>
   280→        <DialogFooter className="flex items-center justify-between">
   281→          {deleting ? (
   282→            <div className="flex items-center gap-2">
   283→              <span className="text-sm text-red-600">Delete event?</span>
   284→              <Button size="sm" variant="destructive" onClick={handleDelete}>Yes</Button>
   285→              <Button size="sm" variant="outline" onClick={() => setDeleting(false)}>Cancel</Button>
   286→            </div>
   287→          ) : (
   288→            <>
   289→              <Button variant="ghost" size="sm" onClick={() => setDeleting(true)} className="text-red-600">Delete</Button>
   290→              <div className="flex gap-2">
   291→                {isUpcoming && <Button size="sm" variant="outline" onClick={onRemind}><Icon as={Bell} boundingBox={false} size={14} />Send reminder</Button>}
   292→                <Button size="sm" onClick={onClose}>Close</Button>
   293→              </div>
   294→            </>
   295→          )}
   296→        </DialogFooter>
   297→      </DialogContent>
   298→    </Dialog>
   299→  );
   300→}

   302→function ReminderDialog({ event, onClose }: { event: EventItem; onClose: () => void }) {
   303→  const [channel, setChannel] = React.useState<"email" | "sms" | "both">("both");
   304→  const [sending, setSending] = React.useState(false);
   305→  const [result, setResult] = React.useState<{ emails: number; sms: number; undeliverable: number; totalMembers: number } | null>(null);
   306→
   307→  async function send() {
   308→    setSending(true);
   309→    const res = await apiRequest<{ emails: number; sms: number; undeliverable: number; totalMembers: number }>("/api/reminders", {
   310→      method: "POST", body: { eventId: event.id, channel },
   311→    });
   312→    setSending(false);
   313→    if (res.ok && res.data) setResult(res.data);
   314→    else toast.error(res.error || "Failed to dispatch");
   315→  }
   316→
   317→  return (
   318→    <Dialog open onOpenChange={onClose}>
   319→      <DialogContent className="max-w-md">
   320→        <DialogHeader>
   321→          <DialogTitle className="flex items-center gap-2"><Icon as={Bell} boundingBox={false} size={18} />Event reminders</DialogTitle>
   322→          <DialogDescription>{event.title} — {formatDateTime(event.startTime)}</DialogDescription>
   323→        </DialogHeader>
   324→        {!result ? (
   325→          <div className="space-y-4 py-2">
   326→            <div className="rounded-sm bg-amber-50 border border-amber-200 px-4 py-3 text-sm text-amber-800">
   327→              Reminders are <strong>simulated</strong> in this environment — they write to the audit log but don't deliver actual messages. (PRD §3.5)
   328→            </div>
   329→            <div>
   330→              <Label className="text-sm font-medium mb-2 block">Channel</Label>
   331→              <Select value={channel} onValueChange={(v: "email" | "sms" | "both") => setChannel(v)}>
   332→                <SelectTrigger><SelectValue /></SelectTrigger>
   333→                <SelectContent>
   334→                  <SelectItem value="both">Email + SMS</SelectItem>
   335→                  <SelectItem value="email">Email only</SelectItem>
   336→                  <SelectItem value="sms">SMS only</SelectItem>
   337→                </SelectContent>
   338→              </Select>
   339→            </div>
   340→          </div>
   341→        ) : (
   342→          <div className="py-4 text-center">
   343→            <div className="w-12 h-12 rounded-full bg-emerald-50 text-emerald-600 flex items-center justify-center mx-auto mb-4">
   344→              <Icon as={Bell} boundingBox={false} size={24} />
   345→            </div>
   346→            <h3 className="text-title text-primary">Reminders dispatched</h3>
   347→            <p className="text-caption mt-1">(Simulated — audit log updated)</p>
   348→            <div className="grid grid-cols-3 gap-4 max-w-xs mx-auto mt-6">
   349→              <div><p className="text-2xl font-semibold text-primary">{result.emails}</p><p className="text-caption">Emails</p></div>
   350→              <div><p className="text-2xl font-semibold text-primary">{result.sms}</p><p className="text-caption">SMS</p></div>
   351→              <div><p className="text-2xl font-semibold text-primary">{result.undeliverable}</p><p className="text-caption">Undeliverable</p></div>
   352→            </div>
   353→          </div>
   354→        )}
   355→        <DialogFooter>
   356→          {!result && <Button size="sm" onClick={send} disabled={sending}>{sending ? "Dispatching…" : "Dispatch reminders"}</Button>}
   357→          {result && <Button size="sm" onClick={onClose}>Done</Button>}
   358→        </DialogFooter>
   359→      </DialogContent>
   360→    </Dialog>
   361→  );
   362→}

   418→function EventAttendanceRecorder({ eventId, onBack }: { eventId: string; onBack: () => void }) {
   419→  const { data, loading, error, refetch } = useFetch<{ attendance: { id: string; memberId: string; status: string; note: string | null }[]; members: { id: string; firstName: string; lastName: string; phone: string | null }[]; event: EventItem | null }>(`/api/attendance?eventId=${eventId}`);
   420→  const [changes, setChanges] = React.useState<Record<string, string>>({});
   421→  const [saving, setSaving] = React.useState(false);
   422→
   423→  if (loading) return <LoadingSpinner label="Loading attendance…" />;
   424→  if (error || !data) return <ErrorState title="Couldn't load attendance" description={error} onRetry={refetch} />;
   425→
   426→  const attendanceMap: Record<string, string> = {};
   427→  for (const a of data.attendance) attendanceMap[a.memberId] = a.status;
   428→
   429→  const getStatus = (memberId: string) => changes[memberId] || attendanceMap[memberId] || "present";
   430→  const dirtyCount = Object.keys(changes).length;
   431→
   432→  async function save() {
   433→    setSaving(true);
   434→    const records = Object.entries(changes).map(([memberId, status]) => ({ memberId, status }));
   435→    const res = await apiRequest("/api/attendance", { method: "POST", body: { eventId, records } });
   436→    setSaving(false);
   437→    if (res.ok) { toast.success(`Saved attendance for ${records.length} members`); setChanges({}); refetch(); }
   438→    else toast.error(res.error || "Failed");
   439→  }
   440→
   441→  const STATUSES = ["present", "absent", "excused", "late"];
   442→
   443→  return (
   444→    <SectionCard>
   445→      <div className="p-4 border-b border-border flex items-center justify-between">
   446→        <div>
   447→          <button onClick={onBack} className="text-sm text-muted-foreground hover:text-primary mb-1 inline-flex items-center gap-1">
   448→            <Icon as={ChevronLeft} boundingBox={false} size={14} />Back to events
   449→          </button>
   450→          <h3 className="text-title text-primary">{data.event?.title}</h3>
   451→          <p className="text-caption">{formatDateTime(data.event?.startTime)}</p>
   452→        </div>
   453→        {dirtyCount > 0 && (
   454→          <Button size="sm" onClick={save} disabled={saving}>
   455→            {saving ? "Saving…" : `Save (${dirtyCount})`}
   456→          </Button>
   457→        )}
   458→      </div>
   459→      <div className="max-h-[60vh] overflow-y-auto scrollbar-thin divide-y divide-border">
   460→        {data.members.map((m) => {
   461→          const status = getStatus(m.id);
   462→          return (
   463→            <div key={m.id} className="p-3 flex items-center justify-between gap-3">
   464→              <div className="flex items-center gap-3 min-w-0">
   465→                <div className="w-8 h-8 rounded-full bg-surface-alt flex items-center justify-center text-xs font-semibold text-primary shrink-0">
   466→                  {(m.firstName[0] || "") + (m.lastName[0] || "")}
   467→                </div>
   468→                <span className="text-sm text-foreground truncate">{m.firstName} {m.lastName}</span>
   469→              </div>
   470→              <div className="flex items-center gap-1 shrink-0">
   471→                {STATUSES.map((s) => (
   472→                  <button
   473→                    key={s}
   474→                    onClick={() => setChanges({ ...changes, [m.id]: s })}
   475→                    className={cn(
   476→                      "px-2.5 py-1 rounded-sm text-[11px] font-medium capitalize border transition-colors",
   477→                      status === s ? statusClass(s) : "border-border text-muted-foreground hover:bg-surface-alt"
   478→                    )}
   479→                  >
   480→                    {s}
   481→                  </button>
   482→                ))}
   483→              </div>
   484→            </div>
   485→          );
   486→        })}
   487→      </div>
   488→      <div className="p-4 border-t border-border flex items-center justify-between text-caption">
   489→        <span>{data.members.length} members · {data.attendance.length} previously recorded</span>
   490→        <span>Present: {data.members.filter((m) => getStatus(m.id) === "present").length}</span>
   491→      </div>
   492→    </SectionCard>
   493→  );
   494→}

   519→function MemberAttendanceHistory({ memberId, onBack }: { memberId: string; onBack: () => void }) {
   520→  const { data, loading, error } = useFetch<{ records: { id: string; status: string; note: string | null; recordedAt: string; event: { id: string; title: string; startTime: string } }[] }>(`/api/attendance?memberId=${memberId}`);
   521→  if (loading) return <LoadingSpinner label="Loading…" />;
   522→  if (error || !data) return <ErrorState title="Couldn't load" description={error} onRetry={() => window.location.reload()} />;
   523→
   524→  const present = data.records.filter((r) => r.status === "present").length;
   525→  const rate = data.records.length > 0 ? Math.round((present / data.records.length) * 100) : 0;
   526→
   527→  return (
   528→    <SectionCard>
   529→      <div className="p-4 border-b border-border">
   530→        <button onClick={onBack} className="text-sm text-muted-foreground hover:text-primary mb-2 inline-flex items-center gap-1">
   531→          <Icon as={ChevronLeft} boundingBox={false} size={14} />Back to members
   532→        </button>
   533→        <div className="grid grid-cols-3 gap-4">
   534→          <div><p className="text-2xl font-semibold text-primary">{data.records.length}</p><p className="text-caption">Total events</p></div>
   535→          <div><p className="text-2xl font-semibold text-primary">{present}</p><p className="text-caption">Present</p></div>
   536→          <div><p className="text-2xl font-semibold text-primary">{rate}%</p><p className="text-caption">Attendance rate</p></div>
   537→        </div>
   538→      </div>
   539→      <div className="divide-y divide-border max-h-[55vh] overflow-y-auto scrollbar-thin">
   540→        {data.records.length === 0 ? <EmptyState title="No attendance recorded" /> : data.records.map((r) => (
   541→          <div key={r.id} className="p-3 flex items-center justify-between">
   542→            <div>
   543→              <p className="text-sm font-medium text-primary">{r.event.title}</p>
