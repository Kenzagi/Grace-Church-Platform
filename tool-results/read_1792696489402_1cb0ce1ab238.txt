   157→// ── Series management (edit / delete) ─────────────────────────────
   158→function ManageSeriesDialog({ onClose, onChanged }: { onClose: () => void; onChanged: () => void }) {
   159→  const { data, loading, error, refetch } = useFetch<{ series: (SermonSeries & { _count?: { media: number } })[] }>(`/api/series`);
   160→  const [editingId, setEditingId] = React.useState<string | null>(null);
   161→  const [deletingId, setDeletingId] = React.useState<string | null>(null);
   162→  const [saving, setSaving] = React.useState(false);
   163→  const [editForm, setEditForm] = React.useState({ name: "", description: "" });
   164→
   165→  const series = data?.series || [];
   166→
   167→  function startEdit(s: SermonSeries) {
   168→    setEditingId(s.id);
   169→    setDeletingId(null);
   170→    setEditForm({ name: s.name, description: s.description || "" });
   171→  }
   172→
   173→  async function handleSave(id: string) {
   174→    if (!editForm.name.trim()) { toast.error("Series name is required"); return; }
   175→    setSaving(true);
   176→    const res = await apiRequest<{ series: SermonSeries }>(`/api/series/${id}`, {
   177→      method: "PATCH",
   178→      body: { name: editForm.name.trim(), description: editForm.description.trim() || null },
   179→    });
   180→    setSaving(false);
   181→    if (res.ok) {
   182→      toast.success("Series updated");
   183→      setEditingId(null);
   184→      refetch();
   185→      onChanged();
   186→    } else {
   187→      toast.error(res.error || "Failed to update series");
   188→    }
   189→  }
   190→
   191→  async function handleDelete(s: SermonSeries & { _count?: { media: number } }) {
   192→    setSaving(true);
   193→    const res = await apiRequest<{ detachedMedia: number }>(`/api/series/${s.id}`, { method: "DELETE" });
   194→    setSaving(false);
   195→    if (res.ok) {
   196→      const n = res.data?.detachedMedia ?? 0;
   197→      toast.success(n > 0 ? `Series deleted — ${n} media item${n !== 1 ? "s" : ""} detached` : "Series deleted");
   198→      setDeletingId(null);
   199→      refetch();
   200→      onChanged();
   201→    } else {
   202→      toast.error(res.error || "Failed to delete series");
   203→    }
   204→  }
   205→
   206→  return (
   207→    <Dialog open onOpenChange={onClose}>
   208→      <DialogContent className="max-w-lg max-h-[85vh] overflow-y-auto scrollbar-thin">
   209→        <DialogHeader>
   210→          <DialogTitle>Manage series</DialogTitle>
   211→          <DialogDescription>Rename or remove teaching series. Media items are never deleted — removing a series simply detaches its media.</DialogDescription>
   212→        </DialogHeader>
   213→        <div className="py-2">
   214→          {loading ? <LoadingSpinner label="Loading series…" />
   215→          : error ? <ErrorState title="Couldn't load series" description={error} onRetry={refetch} />
   216→          : series.length === 0 ? (
   217→            <EmptyState title="No series yet" description="Create a series from the Add media form — it will appear here for editing." />
   218→          ) : (
   219→            <div className="divide-y divide-border rounded-sm border border-border">
   220→              {series.map((s) => {
   221→                const isEditing = editingId === s.id;
   222→                const isDeleting = deletingId === s.id;
   223→                return (
   224→                  <div key={s.id} className="p-3 sm:p-4">
   225→                    {isEditing ? (
   226→                      <div className="space-y-2">
   227→                        <Input value={editForm.name} onChange={(e) => setEditForm({ ...editForm, name: e.target.value })} placeholder="Series name" autoFocus />
   228→                        <Textarea value={editForm.description} onChange={(e) => setEditForm({ ...editForm, description: e.target.value })} placeholder="Description (optional)" rows={2} />
   229→                        <div className="flex justify-end gap-2">
   230→                          <Button size="sm" variant="outline" onClick={() => setEditingId(null)} disabled={saving}>Cancel</Button>
   231→                          <Button size="sm" onClick={() => handleSave(s.id)} disabled={saving || !editForm.name.trim()}>{saving ? "Saving…" : "Save"}</Button>
   232→                        </div>
   233→                      </div>
   234→                    ) : isDeleting ? (
   235→                      <div className="space-y-2">
   236→                        <p className="text-sm text-foreground">
   237→                          Delete <strong>{s.name}</strong>?
   238→                          {s._count && s._count.media > 0 && (
   239→                            <span className="text-muted-foreground"> {s._count.media} media item{s._count.media !== 1 ? "s" : ""} will be detached (not deleted).</span>
   240→                          )}
   241→                        </p>
   242→                        <div className="flex justify-end gap-2">
   243→                          <Button size="sm" variant="outline" onClick={() => setDeletingId(null)} disabled={saving}>Cancel</Button>
   244→                          <Button size="sm" variant="destructive" onClick={() => handleDelete(s)} disabled={saving}>{saving ? "Deleting…" : "Delete series"}</Button>
   245→                        </div>
   246→                      </div>
   247→                    ) : (
   248→                      <div className="flex items-center justify-between gap-3">
   249→                        <div className="min-w-0">
   250→                          <p className="font-medium text-primary truncate">{s.name}</p>
   251→                          <p className="text-caption truncate">
   252→                            {s._count?.media ?? 0} media item{(s._count?.media ?? 0) !== 1 ? "s" : ""}{s.description ? ` · ${s.description}` : ""}
   253→                          </p>
   254→                        </div>
   255→                        <div className="flex items-center gap-1 shrink-0">
   256→                          <Button size="sm" variant="ghost" onClick={() => startEdit(s)} aria-label={`Edit ${s.name}`}>
   257→                            <Icon as={Pencil} boundingBox={false} size={14} />Edit
   258→                          </Button>
   259→                          <Button size="sm" variant="ghost" onClick={() => { setDeletingId(s.id); setEditingId(null); }} className="text-red-600" aria-label={`Delete ${s.name}`}>
   260→                            <Icon as={Trash2} boundingBox={false} size={14} />Delete
   261→                          </Button>
   262→                        </div>
   263→                      </div>
   264→                    )}
   265→                  </div>
   266→                );
   267→              })}
   268→            </div>
   269→          )}
   270→        </div>
   271→        <DialogFooter>
   272→          <Button variant="outline" size="sm" onClick={onClose}>Close</Button>
   273→        </DialogFooter>
   274→      </DialogContent>
   275→    </Dialog>
   276→  );
   277→}

   279→function AddMediaDialog({ series: initialSeries, onClose, onSaved }: { series: SermonSeries[]; onClose: () => void; onSaved: () => void }) {
   280→  const { data: membersData } = useFetch<{ members: Member[] }>(`/api/members?limit=200`);
   281→  const [series, setSeries] = React.useState<SermonSeries[]>(initialSeries);
   282→  const [showInlineSeries, setShowInlineSeries] = React.useState(false);
   283→  const [form, setForm] = React.useState({ title: "", description: "", type: "sermon", url: "", duration: "", speakerId: "", seriesId: "", date: new Date().toISOString().slice(0, 10) });
   284→  const [saving, setSaving] = React.useState(false);
   285→  const members = membersData?.members || [];
   286→
   287→  async function handleSubmit() {
   288→    if (!form.title) { toast.error("Title is required"); return; }
   289→    setSaving(true);
   290→    const res = await apiRequest("/api/media", { method: "POST", body: { ...form, duration: form.duration ? parseInt(form.duration) : null } });
   291→    setSaving(false);
   292→    if (res.ok) { toast.success("Media added"); onSaved(); }
   293→    else toast.error(res.error || "Failed");
   294→  }
   295→
   296→  return (
   297→    <Dialog open onOpenChange={onClose}>
   298→      <DialogContent className="max-w-xl max-h-[90vh] overflow-y-auto scrollbar-thin">
   299→        <DialogHeader>
   300→          <DialogTitle>Add media item</DialogTitle>
   301→          <DialogDescription>Add a new sermon, audio, video, or document to the library.</DialogDescription>
   302→        </DialogHeader>
   303→        <div className="space-y-4 py-4">
   304→          <div><Label className="text-sm font-medium mb-2 block">Title *</Label><Input value={form.title} onChange={(e) => setForm({ ...form, title: e.target.value })} /></div>
   305→          <div><Label className="text-sm font-medium mb-2 block">Description</Label><Textarea value={form.description} onChange={(e) => setForm({ ...form, description: e.target.value })} rows={3} /></div>
   306→          <div className="grid grid-cols-2 gap-4">
   307→            <div>
   308→              <Label className="text-sm font-medium mb-2 block">Type</Label>
   309→              <Select value={form.type} onValueChange={(v) => setForm({ ...form, type: v })}>
   310→                <SelectTrigger><SelectValue /></SelectTrigger>
   311→                <SelectContent>
   312→                  <SelectItem value="sermon">Sermon</SelectItem>
   313→                  <SelectItem value="audio">Audio</SelectItem>
   314→                  <SelectItem value="video">Video</SelectItem>
   315→                  <SelectItem value="document">Document</SelectItem>
   316→                </SelectContent>
   317→              </Select>
   318→            </div>
   319→            <div><Label className="text-sm font-medium mb-2 block">Duration (sec)</Label><Input type="number" value={form.duration} onChange={(e) => setForm({ ...form, duration: e.target.value })} /></div>
   320→            <div><Label className="text-sm font-medium mb-2 block">Speaker</Label>
   321→              <Select value={form.speakerId} onValueChange={(v) => setForm({ ...form, speakerId: v })}>
   322→                <SelectTrigger>
   323→                  <SelectValue placeholder={members.length === 0 ? "No members available" : "Select speaker"} />
   324→                </SelectTrigger>
   325→                <SelectContent>
   326→                  {members.length === 0 && <SelectItem value="_empty" disabled className="text-muted-foreground italic">No members available</SelectItem>}
   327→                  {members.map((m) => <SelectItem key={m.id} value={m.id}>{fullName(m.firstName, m.lastName)}</SelectItem>)}
   328→                </SelectContent>
   329→              </Select>
   330→            </div>
   331→            <div><Label className="text-sm font-medium mb-2 block">Series</Label>
   332→              <Select value={form.seriesId} onValueChange={(v) => setForm({ ...form, seriesId: v })}>
   333→                <SelectTrigger>
   334→                  <SelectValue placeholder={series.length === 0 ? "No series yet" : "No series"} />
   335→                </SelectTrigger>
   336→                <SelectContent>
   337→                  {series.length === 0 && <SelectItem value="_empty" disabled className="text-muted-foreground italic">No series created yet</SelectItem>}

   339→                </SelectContent>

   342→                <button

   355→                    setShowInlineSeries(false);
   356→                  }}
   357→                  onCancel={() => setShowInlineSeries(false)}
   358→                />
   359→              )}
   360→            </div>

   369→      </DialogContent>
   370→    </Dialog>

   430→// ── Submissions review queue ──────────────────────────────────────
   431→export function SubmissionsSection() {
   432→  const [statusFilter, setStatusFilter] = React.useState("all");
   433→  const { data, loading, error, refetch } = useFetch<{ submissions: Submission[] }>(`/api/submissions?status=${statusFilter}`);

   445→          <div className="flex items-center gap-2">
   446→            <Select value={statusFilter} onValueChange={setStatusFilter}>
   447→              <SelectTrigger className="w-36 h-9"><SelectValue /></SelectTrigger>
   448→              <SelectContent>
   449→                <SelectItem value="all">All</SelectItem>
   450→                <SelectItem value="pending">Pending</SelectItem>
   451→                <SelectItem value="approved">Approved</SelectItem>

   458→
   459→      {pending > 0 && statusFilter !== "pending" && (
   460→        <div className="rounded-sm border border-amber-200 bg-amber-50 px-4 py-3 mb-4 flex items-center gap-3">
   461→          <Icon as={Inbox} boundingBox={false} size={18} className="text-amber-600" />

   465→
   466→      <SectionCard>
   467→        {loading ? <LoadingSpinner label="Loading submissions…" />
   468→        : error ? <ErrorState title="Couldn't load submissions" description={error} onRetry={refetch} />
   469→        : submissions.length === 0 ? <EmptyState title="No submissions" description="Member-submitted media will appear here for review." />
   470→        : (
   471→          <div className="divide-y divide-border">
   472→            {submissions.map((s) => (

   474→                <div className="flex items-start gap-3 flex-1 min-w-0">
   475→                  <div className="w-10 h-10 rounded-sm bg-surface-alt flex items-center justify-center shrink-0">
   476→                    <Icon as={TYPE_ICONS[s.mediaItem.type] || FileText} boundingBox={false} size={16} />
   477→                  </div>
   478→                  <div className="min-w-0">
   479→                    <p className="font-medium text-primary truncate">{s.mediaItem.title}</p>
   480→                    <p className="text-caption">
   481→                      by {fullName(s.member.firstName, s.member.lastName)} · {formatDate(s.createdAt)}
   482→                    </p>
   483→                    {s.note && <p className="text-caption mt-1 italic">"{s.note}"</p>}
   484→                  </div>
   485→                </div>
   486→                <div className="flex items-center gap-3 shrink-0">
   487→                  <span className={`inline-flex items-center px-2 py-0.5 rounded-sm border text-xs font-medium capitalize ${statusClass(s.status)}`}>{s.status}</span>
   488→                  {s.status === "pending" && (
   489→                    <Button size="sm" variant="outline" onClick={() => setReviewing(s)}>
   490→                      <Icon as={Check} boundingBox={false} size={14} />Review
   491→                    </Button>
   492→                  )}
   493→                </div>
   494→              </div>
   495→            ))}
   496→          </div>
   497→        )}
   498→      </SectionCard>
   499→
   500→      {reviewing && <ReviewDialog submission={reviewing} onClose={() => setReviewing(null)} onDone={() => { refetch(); setReviewing(null); }} />}
   501→    </div>
   502→  );
   503→}

   505→function ReviewDialog({ submission, onClose, onDone }: { submission: Submission; onClose: () => void; onDone: () => void }) {
   506→  const [note, setNote] = React.useState("");
   507→  const [acting, setActing] = React.useState(false);
   508→
   509→  async function act(status: "approved" | "rejected") {
   510→    setActing(true);
   511→    const res = await apiRequest(`/api/submissions/${submission.id}`, { method: "PATCH", body: { status, note } });
   512→    setActing(false);
   513→    if (res.ok) {
   514→      toast.success(status === "approved" ? "Submission approved — media published" : "Submission rejected");
   515→      onDone();
   516→    } else toast.error(res.error || "Failed");
   517→  }
   518→
   519→  return (
   520→    <Dialog open onOpenChange={onClose}>
   521→      <DialogContent className="max-w-lg">
   522→        <DialogHeader>
   523→          <DialogTitle>Review submission</DialogTitle>
   524→          <DialogDescription>{submission.mediaItem.title} — by {fullName(submission.member.firstName, submission.member.lastName)}</DialogDescription>
   525→        </DialogHeader>
   526→        <div className="space-y-4 py-2">
   527→          <div className="rounded-sm bg-surface-alt p-4">
   528→            <p className="text-sm text-foreground">{submission.mediaItem.description || "No description provided."}</p>
   529→            {submission.mediaItem.url && (
   530→              <a href={submission.mediaItem.url} target="_blank" rel="noreferrer" className="text-sm text-primary hover:underline mt-2 inline-flex items-center gap-1">
   531→                <Icon as={Play} boundingBox={false} size={14} />Listen / view
   532→              </a>
   533→            )}
   534→          </div>
   535→          <div>
   536→            <Label className="text-sm font-medium mb-2 block">Review note (optional)</Label>
   537→            <Textarea value={note} onChange={(e) => setNote(e.target.value)} rows={3} placeholder="Add feedback for the submitter…" />
   538→          </div>
   539→        </div>
   540→        <DialogFooter className="flex items-center justify-between">
   541→          <Button variant="destructive" size="sm" onClick={() => act("rejected")} disabled={acting}>
   542→            <Icon as={X} boundingBox={false} size={14} />Reject
   543→          </Button>
   544→          <Button size="sm" onClick={() => act("approved")} disabled={acting}>
   545→            <Icon as={Check} boundingBox={false} size={14} />Approve & publish
   546→          </Button>
   547→        </DialogFooter>
   548→      </DialogContent>
   549→    </Dialog>
   550→  );
   551→}

   553→function useDebounce<T>(value: T, delay: number): T {
   554→  const [d, setD] = React.useState(value);
   555→  React.useEffect(() => { const t = setTimeout(() => setD(value), delay); return () => clearTimeout(t); }, [value, delay]);
   556→  return d;
   557→}

   559→// ── Inline series creation (used inside Add Media form) ────
   560→function InlineSeriesCreate({ onCreated, onCancel }: { onCreated: (s: SermonSeries) => void; onCancel: () => void }) {
   561→  const [name, setName] = React.useState("");
   562→  const [saving, setSaving] = React.useState(false);
   563→
   564→  async function handleCreate() {
   565→    if (!name.trim()) return;
   566→    setSaving(true);
   567→    const res = await apiRequest<{ series: SermonSeries }>("/api/series", { method: "POST", body: { name: name.trim() } });
   568→    setSaving(false);
   569→    if (res.ok && res.data?.series) {
   570→      toast.success(`Series "${name.trim()}" created`);
   571→      onCreated(res.data.series);
   572→    } else {
   573→      toast.error(res.error || "Failed to create series");
   574→    }
   575→  }
   576→
   577→  return (
   578→    <div className="rounded-sm border border-primary/30 p-3 bg-surface-alt space-y-2 mt-1.5">
   579→      <p className="text-xs font-medium text-primary">Quick create series</p>
   580→      <Input value={name} onChange={(e) => setName(e.target.value)} placeholder="e.g. Fruit of the Spirit" autoFocus />
   581→      <div className="flex justify-end gap-2">
   582→        <Button size="sm" variant="outline" onClick={onCancel}>Cancel</Button>
   583→        <Button size="sm" onClick={handleCreate} disabled={saving || !name.trim()}>{saving ? "Creating…" : "Create"}</Button>
   584→      </div>
   585→    </div>
   586→  );
   587→}
