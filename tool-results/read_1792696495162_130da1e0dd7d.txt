   158→   158→function ManageSeriesDialog({ onClose, onChanged }: { onClose: () => void; onChanged: () => void }) {
   159→   159→  const { data, loading, error, refetch } = useFetch<{ series: (SermonSeries & { _count?: { media: number } })[] }>(`/api/series`);
   160→   160→  const [editingId, setEditingId] = React.useState<string | null>(null);
   161→   161→  const [deletingId, setDeletingId] = React.useState<string | null>(null);
   162→   162→  const [saving, setSaving] = React.useState(false);
   163→   163→  const [editForm, setEditForm] = React.useState({ name: "", description: "" });
   164→   164→
   165→   165→  const series = data?.series || [];
   166→   166→
   167→   167→  function startEdit(s: SermonSeries) {
   168→   168→    setEditingId(s.id);
   169→   169→    setDeletingId(null);
   170→   170→    setEditForm({ name: s.name, description: s.description || "" });
   171→   171→  }
   172→   172→
   173→   173→  async function handleSave(id: string) {
   174→   174→    if (!editForm.name.trim()) { toast.error("Series name is required"); return; }
   175→   175→    setSaving(true);
   176→   176→    const res = await apiRequest<{ series: SermonSeries }>(`/api/series/${id}`, {
   177→   177→      method: "PATCH",
   178→   178→      body: { name: editForm.name.trim(), description: editForm.description.trim() || null },
   179→   179→    });
   180→   180→    setSaving(false);
   181→   181→    if (res.ok) {
   182→   182→      toast.success("Series updated");
   183→   183→      setEditingId(null);
   184→   184→      refetch();
   185→   185→      onChanged();
   186→   186→    } else {
   187→   187→      toast.error(res.error || "Failed to update series");
   188→   188→    }
   189→   189→  }
   190→   190→
   191→   191→  async function handleDelete(s: SermonSeries & { _count?: { media: number } }) {
   192→   192→    setSaving(true);
   193→   193→    const res = await apiRequest<{ detachedMedia: number }>(`/api/series/${s.id}`, { method: "DELETE" });
   194→   194→    setSaving(false);
   195→   195→    if (res.ok) {
   196→   196→      const n = res.data?.detachedMedia ?? 0;
   197→   197→      toast.success(n > 0 ? `Series deleted — ${n} media item${n !== 1 ? "s" : ""} detached` : "Series deleted");
   198→   198→      setDeletingId(null);
   199→   199→      refetch();
   200→   200→      onChanged();
   201→   201→    } else {
   202→   202→      toast.error(res.error || "Failed to delete series");
   203→   203→    }
   204→   204→  }
   205→   205→
   206→   206→  return (
   207→   207→    <Dialog open onOpenChange={onClose}>
   208→   208→      <DialogContent className="max-w-lg max-h-[85vh] overflow-y-auto scrollbar-thin">
   209→   209→        <DialogHeader>
   210→   210→          <DialogTitle>Manage series</DialogTitle>
   211→   211→          <DialogDescription>Rename or remove teaching series. Media items are never deleted — removing a series simply detaches its media.</DialogDescription>
   212→   212→        </DialogHeader>
   213→   213→        <div className="py-2">
   214→   214→          {loading ? <LoadingSpinner label="Loading series…" />
   215→   215→          : error ? <ErrorState title="Couldn't load series" description={error} onRetry={refetch} />
   216→   216→          : series.length === 0 ? (
   217→   217→            <EmptyState title="No series yet" description="Create a series from the Add media form — it will appear here for editing." />
   218→   218→          ) : (
   219→   219→            <div className="divide-y divide-border rounded-sm border border-border">
   220→   220→              {series.map((s) => {
   221→   221→                const isEditing = editingId === s.id;
   222→   222→                const isDeleting = deletingId === s.id;
   223→   223→                return (
   224→   224→                  <div key={s.id} className="p-3 sm:p-4">
   225→   225→                    {isEditing ? (
   226→   226→                      <div className="space-y-2">
   227→   227→                        <Input value={editForm.name} onChange={(e) => setEditForm({ ...editForm, name: e.target.value })} placeholder="Series name" autoFocus />
   228→   228→                        <Textarea value={editForm.description} onChange={(e) => setEditForm({ ...editForm, description: e.target.value })} placeholder="Description (optional)" rows={2} />
   229→   229→                        <div className="flex justify-end gap-2">
   230→   230→                          <Button size="sm" variant="outline" onClick={() => setEditingId(null)} disabled={saving}>Cancel</Button>
   231→   231→                          <Button size="sm" onClick={() => handleSave(s.id)} disabled={saving || !editForm.name.trim()}>{saving ? "Saving…" : "Save"}</Button>
   232→   232→                        </div>
   233→   233→                      </div>
   234→   234→                    ) : isDeleting ? (
   235→   235→                      <div className="space-y-2">
   236→   236→                        <p className="text-sm text-foreground">
   237→   237→                          Delete <strong>{s.name}</strong>?
   238→   238→                          {s._count && s._count.media > 0 && (
   239→   239→                            <span className="text-muted-foreground"> {s._count.media} media item{s._count.media !== 1 ? "s" : ""} will be detached (not deleted).</span>
   240→   240→                          )}
   241→   241→                        </p>
   242→   242→                        <div className="flex justify-end gap-2">
   243→   243→                          <Button size="sm" variant="outline" onClick={() => setDeletingId(null)} disabled={saving}>Cancel</Button>
   244→   244→                          <Button size="sm" variant="destructive" onClick={() => handleDelete(s)} disabled={saving}>{saving ? "Deleting…" : "Delete series"}</Button>
   245→   245→                        </div>
   246→   246→                      </div>
   247→   247→                    ) : (
   248→   248→                      <div className="flex items-center justify-between gap-3">
   249→   249→                        <div className="min-w-0">
   250→   250→                          <p className="font-medium text-primary truncate">{s.name}</p>
   251→   251→                          <p className="text-caption truncate">
   252→   252→                            {s._count?.media ?? 0} media item{(s._count?.media ?? 0) !== 1 ? "s" : ""}{s.description ? ` · ${s.description}` : ""}
   253→   253→                          </p>
   254→   254→                        </div>
   255→   255→                        <div className="flex items-center gap-1 shrink-0">
   256→   256→                          <Button size="sm" variant="ghost" onClick={() => startEdit(s)} aria-label={`Edit ${s.name}`}>
   257→   257→                            <Icon as={Pencil} boundingBox={false} size={14} />Edit
   258→   258→                          </Button>
   259→   259→                          <Button size="sm" variant="ghost" onClick={() => { setDeletingId(s.id); setEditingId(null); }} className="text-red-600" aria-label={`Delete ${s.name}`}>
   260→   260→                            <Icon as={Trash2} boundingBox={false} size={14} />Delete
   261→   261→                          </Button>
   262→   262→                        </div>
   263→   263→                      </div>
   264→   264→                    )}
   265→   265→                  </div>
   266→   266→                );
   267→   267→              })}
   268→   268→            </div>
   269→   269→          )}
   270→   270→        </div>
   271→   271→        <DialogFooter>
   272→   272→          <Button variant="outline" size="sm" onClick={onClose}>Close</Button>
   273→   273→        </DialogFooter>
   274→   274→      </DialogContent>
   275→   275→    </Dialog>
   276→   276→  );
   277→   277→}

   505→   505→function ReviewDialog({ submission, onClose, onDone }: { submission: Submission; onClose: () => void; onDone: () => void }) {
   506→   506→  const [note, setNote] = React.useState("");
   507→   507→  const [acting, setActing] = React.useState(false);
   508→   508→
   509→   509→  async function act(status: "approved" | "rejected") {
   510→   510→    setActing(true);
   511→   511→    const res = await apiRequest(`/api/submissions/${submission.id}`, { method: "PATCH", body: { status, note } });
   512→   512→    setActing(false);
   513→   513→    if (res.ok) {
   514→   514→      toast.success(status === "approved" ? "Submission approved — media published" : "Submission rejected");
   515→   515→      onDone();
   516→   516→    } else toast.error(res.error || "Failed");
   517→   517→  }
   518→   518→
   519→   519→  return (
   520→   520→    <Dialog open onOpenChange={onClose}>
   521→   521→      <DialogContent className="max-w-lg">
   522→   522→        <DialogHeader>
   523→   523→          <DialogTitle>Review submission</DialogTitle>
   524→   524→          <DialogDescription>{submission.mediaItem.title} — by {fullName(submission.member.firstName, submission.member.lastName)}</DialogDescription>
   525→   525→        </DialogHeader>
   526→   526→        <div className="space-y-4 py-2">
   527→   527→          <div className="rounded-sm bg-surface-alt p-4">
   528→   528→            <p className="text-sm text-foreground">{submission.mediaItem.description || "No description provided."}</p>
   529→   529→            {submission.mediaItem.url && (
   530→   530→              <a href={submission.mediaItem.url} target="_blank" rel="noreferrer" className="text-sm text-primary hover:underline mt-2 inline-flex items-center gap-1">
   531→   531→                <Icon as={Play} boundingBox={false} size={14} />Listen / view
   532→   532→              </a>
   533→   533→            )}
   534→   534→          </div>
   535→   535→          <div>
   536→   536→            <Label className="text-sm font-medium mb-2 block">Review note (optional)</Label>
   537→   537→            <Textarea value={note} onChange={(e) => setNote(e.target.value)} rows={3} placeholder="Add feedback for the submitter…" />
   538→   538→          </div>
   539→   539→        </div>
   540→   540→        <DialogFooter className="flex items-center justify-between">
   541→   541→          <Button variant="destructive" size="sm" onClick={() => act("rejected")} disabled={acting}>
   542→   542→            <Icon as={X} boundingBox={false} size={14} />Reject
   543→   543→          </Button>
   544→   544→          <Button size="sm" onClick={() => act("approved")} disabled={acting}>
   545→   545→            <Icon as={Check} boundingBox={false} size={14} />Approve & publish
   546→   546→          </Button>
   547→   547→        </DialogFooter>
   548→   548→      </DialogContent>
   549→   549→    </Dialog>
   550→   550→  );
   551→   551→}

   553→   553→function useDebounce<T>(value: T, delay: number): T {
   554→   554→  const [d, setD] = React.useState(value);
   555→   555→  React.useEffect(() => { const t = setTimeout(() => setD(value), delay); return () => clearTimeout(t); }, [value, delay]);
   556→   556→  return d;
   557→   557→}

   560→   560→function InlineSeriesCreate({ onCreated, onCancel }: { onCreated: (s: SermonSeries) => void; onCancel: () => void }) {

   563→   563→
   564→   564→  async function handleCreate() {

   567→   567→    const res = await apiRequest<{ series: SermonSeries }>("/api/series", { method: "POST", body: { name: name.trim() } });

   569→   569→    if (res.ok && res.data?.series) {
   570→   570→      toast.success(`Series "${name.trim()}" created`);

   575→   575→  }

   582→   582→        <Button size="sm" variant="outline" onClick={onCancel}>Cancel</Button>
